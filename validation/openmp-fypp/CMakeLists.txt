find_program(FYPP_EXECUTABLE NAMES fypp HINTS "${WORKSPACE_ROOT}/fypp/bin" "${WORKSPACE_ROOT}/../fypp/bin" "${WORKSPACE_ROOT}/toolchains/fypp-venv/bin")

  message(FATAL_ERROR "Fypp release 3.3 is required. Provide FYPP_EXECUTABLE or the local release checkout.")

execute_process(COMMAND "${FYPP_EXECUTABLE}" --version RESULT_VARIABLE FYPP_STATUS
  OUTPUT_VARIABLE FYPP_VERSION ERROR_VARIABLE FYPP_ERROR)
if(NOT FYPP_STATUS EQUAL 0 OR NOT "${FYPP_VERSION}${FYPP_ERROR}" MATCHES "(^|[^0-9])3\\.3([^0-9.]|$)")
  message(FATAL_ERROR "Expected Fypp release 3.3; found ${FYPP_VERSION}${FYPP_ERROR}")
endif()
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GENERATED}")
function(generate_fypp TEMPLATE NAME BACKEND)
  add_custom_command(OUTPUT "${GENERATED}/${NAME}.f90"
    COMMAND "${FYPP_EXECUTABLE}" -I "${CMAKE_CURRENT_SOURCE_DIR}" "-SBACKEND=${BACKEND}"
      --depfile "${GENERATED}/${NAME}.f90.d"
      "${CMAKE_CURRENT_SOURCE_DIR}/${TEMPLATE}.fypp" "${GENERATED}/${NAME}.f90"
    DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/${TEMPLATE}.fypp" "${FYPP_EXECUTABLE}"
    DEPFILE "${GENERATED}/${NAME}.f90.d" VERBATIM)
endfunction()
generate_fypp(core probe_core "${GENERATED_BACKEND}")
generate_fypp(probe probe "${GENERATED_BACKEND}")
generate_fypp(snapshot snapshot "${GENERATED_BACKEND}")
generate_fypp(snapshot snapshot_cpu serial)

# All targets that execute the selected backend receive compile AND link
# requirements transitively. The CPU reference deliberately does not inherit it.
add_library(probe_execution INTERFACE)
if(NOT GENERATED_BACKEND STREQUAL "serial")
  find_package(OpenMP REQUIRED COMPONENTS Fortran)
  target_link_libraries(probe_execution INTERFACE OpenMP::OpenMP_Fortran)
endif()
if(GENERATED_BACKEND STREQUAL "target_gpu")
  if(NOT PROBE_OFFLOAD_TARGET MATCHES "^(nvptx-none|amdgcn-amdhsa)$")
    message(FATAL_ERROR "Choose a supported GNU offload target")
  endif()
  target_compile_options(probe_execution INTERFACE "-foffload=${PROBE_OFFLOAD_TARGET}")
  # Linking libgomp alone does not activate the GCC offload start/end tables.
  # The final Fortran driver invocation also needs -fopenmp.
  target_link_options(probe_execution INTERFACE -fopenmp "-foffload=${PROBE_OFFLOAD_TARGET}")
  if(PROBE_BACKEND STREQUAL "nvidia_gcc12")
    # The real-valued Swift power calls pow on the device. Linking host libm
    # does not also link the NVPTX math library into the offload image.
    target_link_options(probe_execution INTERFACE "-foffload-options=nvptx-none=-lm")
    # GCC's lowering of the instrumented reduction also needs device atomics.
    target_link_options(probe_execution INTERFACE "-foffload-options=nvptx-none=-latomic")
  endif()
  if(GCC_OFFLOAD_PREFIX)
    execute_process(COMMAND "${CMAKE_Fortran_COMPILER}" -dumpmachine OUTPUT_VARIABLE GCC_MACHINE OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    string(REGEX MATCH "^[0-9]+" GCC_MAJOR "${CMAKE_Fortran_COMPILER_VERSION}")
    set(OFFLOAD_SEARCH "${GCC_OFFLOAD_PREFIX}/usr/lib/gcc/${GCC_MACHINE}/${GCC_MAJOR}/")
    target_compile_options(probe_execution INTERFACE "-B${OFFLOAD_SEARCH}")
    target_link_options(probe_execution INTERFACE "-B${OFFLOAD_SEARCH}")
  endif()
  if(PROBE_GPU_ARCH)
    set(ARCH_FLAGS "-foffload-options=${PROBE_OFFLOAD_TARGET}=-march=${PROBE_GPU_ARCH}")
    target_compile_options(probe_execution INTERFACE "${ARCH_FLAGS}")
    target_link_options(probe_execution INTERFACE "${ARCH_FLAGS}")
  endif()
elseif(NOT GENERATED_BACKEND STREQUAL "serial")
  target_compile_options(probe_execution INTERFACE -foffload=disable)
  target_link_options(probe_execution INTERFACE -foffload=disable)
endif()
function(add_core TARGET MODULES)
  file(MAKE_DIRECTORY "${MODULES}")
  add_library(${TARGET} STATIC "${GENERATED}/probe_core.f90")
  set_target_properties(${TARGET} PROPERTIES Fortran_MODULE_DIRECTORY "${MODULES}")
  target_include_directories(${TARGET} PUBLIC "${MODULES}" PRIVATE "${MUSCLE_SOURCE_DIR}/src/math")
endfunction()

target_link_libraries(probe_core_selected PUBLIC probe_execution)
add_core(probe_core_cpu "${CMAKE_CURRENT_BINARY_DIR}/modules/cpu")
add_executable(probe_fypp "${GENERATED}/probe.f90")
target_link_libraries(probe_fypp PRIVATE probe_core_selected)

add_executable(probe_snapshot_cpu "${GENERATED}/snapshot_cpu.f90")
target_link_libraries(probe_snapshot_cpu PRIVATE probe_core_cpu)
